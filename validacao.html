// ============================================================ // Apps Script — Backend da planilha de VALIDAÇÃO DE MÉTODOS // Mesma lógica/contrato do painel principal, pra funcionar com // o mesmo tipo de HTML. // ============================================================ const NOME_ABA = 'Entradas'; function getSheet(){ return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(NOME_ABA); } // Lê todas as entradas e devolve como JSON function doGet(e){ const sheet = getSheet(); const dados = sheet.getDataRange().getValues(); const cabecalho = dados[0]; const linhas = dados.slice(1); const resultado = linhas .filter(linha => linha[0] !== '') // ignora linhas vazias .map(linha => { const obj = {}; cabecalho.forEach((coluna, i) => { let valor = linha[i]; if (valor instanceof Date) { valor = Utilities.formatDate(valor, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } obj[coluna] = valor; }); return obj; }); return ContentService.createTextOutput(JSON.stringify(resultado)) .setMimeType(ContentService.MimeType.JSON); } // Recebe uma nova entrada, uma atualização ou uma remoção function doPost(e){ const sheet = getSheet(); const corpo = JSON.parse(e.postData.contents); const cabecalho = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; if (corpo.acao === 'remover'){ removerLinha(sheet, cabecalho, corpo.id); } else if (corpo.acao === 'atualizarTudo'){ atualizarLinha(sheet, cabecalho, corpo); } else { adicionarLinha(sheet, cabecalho, corpo); } return ContentService.createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } function mapaCorpo(corpo){ return { 'ID': corpo.id || new Date().getTime(), 'Data': corpo.data || '', 'Casa': corpo.casa || '', 'Campeonato': corpo.campeonato || '', 'Jogo': corpo.jogo || '', 'Tempo': corpo.tempo || '', 'Tipo': corpo.tipo || '', 'Operação': corpo.operacao || '', 'Odd': corpo.odd || '', 'Stake': corpo.stake || '', 'Responsabilidade': corpo.responsabilidade || '', 'Resultado': corpo.resultado || '', 'Lucro': corpo.lucro || 0, 'Observações': corpo.obs || '' }; } function adicionarLinha(sheet, cabecalho, corpo){ const mapa = mapaCorpo(corpo); const linha = cabecalho.map(col => (mapa[col] !== undefined ? mapa[col] : '')); sheet.appendRow(linha); } function atualizarLinha(sheet, cabecalho, corpo){ const mapa = mapaCorpo(corpo); const colId = cabecalho.indexOf('ID'); const dados = sheet.getDataRange().getValues(); for (let i = 1; i < dados.length; i++){ if (String(dados[i][colId]) === String(corpo.id)){ const linha = cabecalho.map((col, idx) => idx === colId ? dados[i][colId] : (mapa[col] !== undefined ? mapa[col] : '') ); sheet.getRange(i + 1, 1, 1, cabecalho.length).setValues([linha]); break; } } } function removerLinha(sheet, cabecalho, id){ const colId = cabecalho.indexOf('ID'); const dados = sheet.getDataRange().getValues(); for (let i = 1; i < dados.length; i++){ if (String(dados[i][colId]) === String(id)){ sheet.deleteRow(i + 1); break; } } }
